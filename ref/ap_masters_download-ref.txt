[Collection: ApMasters]
	List Name: ##ApMastersDownloadTables
;	List Name: "account_type,voucher_type,account"

	
[Function: APMastersDownload]
	Variable: ApMastersDownloadTables : String
	Variable: SVHttpCallCounter: String	: "0"
		
	00000: Query Box: "Do you want to download masters:\n voucher_type, account_type" : Yes: No
	00100: If: $$LastResult			
	00200: 	START BLOCK
	00300: 	 Set Object: (Company, ##SVCurrentCompany).
	00400: 	 LIST FILL : VoucherTypeMapList 	: APVoucherTypes	: $APObjId 		: $APTallyName
	00500: 	 LIST FILL : GroupMapList 			  : APGroups 				: $APObjId 		: $APTallyName
	00600: 	END BLOCK
	00700:	 Set: ApMastersDownloadTables: "account_type,voucher_type"	
	00800:   Set: SVHttpCallCounter: "1"						;; calling http request two times using modifing call counter
	00900: 	 Walk Collection: ApMastersDownload DataSource		 
	01000: 	  SET TARGET	: (Company, ##SVCurrentCompany).
	01100: 	 End Walk
	01200:	 Set: ApMastersDownloadTables: "account"
	01300:   Set: SVHttpCallCounter: "2"						;; calling http request two times using modifing call counter
	01400: 	 Walk Collection: ApMastersDownload DataSource
	01500: 	  SET TARGET	: (Company, ##SVCurrentCompany).
	01600: 	 End Walk
	01700: End If
