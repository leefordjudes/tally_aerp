;;===== Functions ==================================================================

[Function: APLoginSubmit]
	Parameter: ApUsername: String
	Parameter: ApPassword: String
	
	00100: If: $$IsEmpty:$ApHost Or $$IsEmpty:$ApBookBegin
	00200: 	Msg Box: "Missing Configuration": "Host/BookBegin not filled"
	00300: End If

	00400: Walk Collection: ApLogin DataSource
	00500: 	Set Target: (Company, ##SvCurrentCompany).
	00600:  If: $status = "failure"
	00700:  	Reset Value: ApAuthToken
	00800: 		Msg Box: "Response": "Login Failed\n\n" + $error
	00900: 	End If	
	01000: 	If: $status = "success"
	01100:  	Set Value: ApAuthToken : $data	
	01200: 		Msg Box: "Response": "Login Success"
;	01200: 		Msg Box: "Response": "Login Success\n\n Token Received \n\n" + $data
	01300: 	End If
	01400:  Alter Target
	01500: End Walk
